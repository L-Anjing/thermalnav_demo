cmake_minimum_required(VERSION 3.16)
project(thermalnav_demo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs objdetect dnn calib3d)
find_package(nlohmann_json REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_package(octomap REQUIRED)
add_library(thermalnav_core STATIC src/core.cpp src/renderer.cpp src/perception.cpp src/stereo_frontend.cpp src/occupancy_fusion.cpp)
target_include_directories(thermalnav_core PUBLIC include ${OpenCV_INCLUDE_DIRS} ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(thermalnav_core PUBLIC ${OpenCV_LIBS} nlohmann_json::nlohmann_json CURL::libcurl Threads::Threads ${OCTOMAP_LIBRARIES})
add_executable(thermalnav src/node.cpp)
target_link_libraries(thermalnav thermalnav_core)
ament_target_dependencies(thermalnav rclcpp sensor_msgs nav_msgs geometry_msgs visualization_msgs std_msgs std_srvs tf2_ros cv_bridge)
add_executable(thermalnav_tools src/tools.cpp)
target_link_libraries(thermalnav_tools thermalnav_core)
ament_target_dependencies(thermalnav_tools rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs tf2_ros cv_bridge)
add_executable(thermalnav_stereo src/stereo_node.cpp)
target_link_libraries(thermalnav_stereo thermalnav_core)
ament_target_dependencies(thermalnav_stereo rclcpp sensor_msgs std_msgs tf2_ros cv_bridge)
add_executable(thermalnav_evaluate src/fusion_evaluation.cpp)
target_link_libraries(thermalnav_evaluate thermalnav_core)
ament_target_dependencies(thermalnav_evaluate rclcpp sensor_msgs nav_msgs std_msgs tf2_ros cv_bridge)
install(TARGETS thermalnav thermalnav_tools thermalnav_stereo thermalnav_evaluate DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
# Remove only files installed by the former Python package during an upgrade.
install(CODE [[
  set(_prefix "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}")
  file(GLOB _python_package
    "${_prefix}/local/lib/python*/dist-packages/thermalnav"
    "${_prefix}/local/lib/python*/dist-packages/thermalnav-*.egg-info")
  if(_python_package)
    file(REMOVE_RECURSE ${_python_package})
  endif()
  file(REMOVE
    "${_prefix}/share/thermalnav_demo/environment/pythonpath.sh"
    "${_prefix}/share/thermalnav_demo/environment/pythonpath.dsv")
]])
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core tests/test_core.cpp)
  target_link_libraries(test_core thermalnav_core)
  ament_add_gtest(test_stereo_fusion tests/test_stereo_fusion.cpp)
  target_link_libraries(test_stereo_fusion thermalnav_core)
  ament_target_dependencies(test_stereo_fusion sensor_msgs std_msgs)
  ament_add_gtest(test_ros_runtime tests/test_ros_runtime.cpp SKIP_LINKING_MAIN_LIBRARIES TIMEOUT 180)
  target_link_libraries(test_ros_runtime thermalnav_core)
  ament_target_dependencies(test_ros_runtime rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs tf2_ros)
  target_compile_definitions(test_ros_runtime PRIVATE THERMALNAV_BINARY="$<TARGET_FILE:thermalnav>" THERMALNAV_TOOLS_BINARY="$<TARGET_FILE:thermalnav_tools>" THERMALNAV_STEREO_BINARY="$<TARGET_FILE:thermalnav_stereo>" THERMALNAV_EVALUATION_BINARY="$<TARGET_FILE:thermalnav_evaluate>")
  add_dependencies(test_ros_runtime thermalnav thermalnav_tools thermalnav_stereo thermalnav_evaluate)
  set_tests_properties(test_ros_runtime PROPERTIES LABELS integration)
endif()
ament_package()
